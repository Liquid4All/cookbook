cmake_minimum_required(VERSION 3.31.6)
project(liquid-android-runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
    set(GGML_OPENMP ON CACHE BOOL "" FORCE)
else()
    set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
endif()
if(LIQUID_RUNTIME STREQUAL "vision" OR LIQUID_RUNTIME STREQUAL "all")
    set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
else()
    set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(${LLAMA_SRC} build-llama)
foreach(kind IN ITEMS text agent vision)
    if(LIQUID_RUNTIME STREQUAL kind OR LIQUID_RUNTIME STREQUAL "all")
        add_library(liquid-${kind}-jni SHARED ${kind}_jni.cpp)
        target_include_directories(liquid-${kind}-jni PRIVATE ${LLAMA_SRC}/common ${LLAMA_SRC}/tools/mtmd)
        target_link_libraries(liquid-${kind}-jni llama llama-common android log)
        if(kind STREQUAL "vision")
            target_link_libraries(liquid-${kind}-jni mtmd)
        endif()
    endif()
endforeach()
