cmake_minimum_required(VERSION 3.31.6)
project(vision_demo LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_SYSTEM_ARCH "ARM")
    set(GGML_CPU_KLEIDIAI ON)
    set(GGML_OPENMP ON)
else()
    set(GGML_SYSTEM_ARCH "x86")
    set(GGML_CPU_KLEIDIAI OFF)
    set(GGML_OPENMP OFF)
endif()
set(LLAMA_SRC ${CMAKE_CURRENT_LIST_DIR}/../../../../llama.cpp)
add_subdirectory(${LLAMA_SRC} build-llama)
add_library(vision-demo SHARED vision.cpp)
target_include_directories(vision-demo PRIVATE ${LLAMA_SRC}/common ${LLAMA_SRC}/include ${LLAMA_SRC}/tools/mtmd ${LLAMA_SRC}/ggml/include)
target_link_libraries(vision-demo llama llama-common mtmd android log)
